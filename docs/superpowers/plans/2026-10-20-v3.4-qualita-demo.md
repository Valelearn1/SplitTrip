    /** Un commento tolto lascia i suoi a capo: il numero di riga nell'errore resta quello del file. */
    static String stripComments(String src) {
        String noBlocks = Pattern.compile("(?s)/\\*.*?\\*/").matcher(src).replaceAll(m -> m.group().replaceAll("[^\\n]", ""));
        return noBlocks.replaceAll("(?m)(^|[^:])//.*$", "$1");
    }
```

/** Un quadrato 1×1 per ogni modulo scuro: un solo <path>, niente migliaia di <rect>. */
export const qrPath = (data: boolean[][]) =>
  data.flatMap((row, y) => row.map((on, x) => (on ? `M${x} ${y}h1v1h-1z` : ''))).join('')
```

`FE/src/features/trips/InviteQr.tsx`:

```tsx
import { useEffect, useState } from 'react'
import { qrPath } from './qrPath'

/** D-67: il QR si calcola nel telefono e uqr arriva solo quando serve. Sempre scuro su bianco, anche nel tema scuro. */
export function InviteQr({ link, label }: { link: string; label: string }) {
  const [data, setData] = useState<boolean[][] | null>(null)
  const [failed, setFailed] = useState(false)
  useEffect(() => {
    let alive = true
    import('uqr')
      .then(({ encode }) => { if (alive) setData(encode(link, { ecc: 'M', border: 4 }).data) })
      .catch(() => { if (alive) setFailed(true) })
    return () => { alive = false }
  }, [link])
  if (failed) return <p className="muted">Usa il codice qui sotto</p>
  if (!data) return <div className="qr qr-placeholder" aria-hidden="true" />
  return (
    <svg className="qr" role="img" aria-label={label} viewBox={`0 0 ${data.length} ${data.length}`} shapeRendering="crispEdges">
      <rect width={data.length} height={data.length} fill="#ffffff" />
      <path fill="#141c26" d={qrPath(data)} />
    </svg>
  )
}
```

```yaml
# D-66: a ogni push su main girano i test del backend, del frontend e il giro end-to-end
name: CI

on:
  push:
    branches: [main]
  workflow_dispatch:

jobs:
  backend:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v5
      - uses: actions/setup-java@v5
        with:
          distribution: temurin
          java-version: '25'
          cache: maven
      - name: Test (Testcontainers usa il Docker della macchina)
        working-directory: BE
        run: sh mvnw -B test

  frontend:
    runs-on: ubuntu-latest
    defaults:
      run:
        working-directory: FE
    steps:
      - uses: actions/checkout@v5
      - uses: actions/setup-node@v5
        with:
          node-version: '24'
          cache: npm
          cache-dependency-path: FE/package-lock.json
      - run: npm ci
      - run: npm run lint
      - run: npm test
      - run: npm run build

  e2e:
    needs: [backend, frontend]
    runs-on: ubuntu-latest
    services:
      postgres:
        image: postgres:16-alpine
        env:
          POSTGRES_DB: splittrip_e2e
          POSTGRES_USER: splittrip
          POSTGRES_PASSWORD: splittrip
        ports:
          - 5433:5432
        options: >-
          --health-cmd "pg_isready -U splittrip"
          --health-interval 5s --health-timeout 5s --health-retries 10
    defaults:
      run:
        working-directory: FE
    steps:
      - uses: actions/checkout@v5
      - uses: actions/setup-java@v5
        with:
          distribution: temurin
          java-version: '25'
          cache: maven
      - uses: actions/setup-node@v5
        with:
          node-version: '24'
          cache: npm
          cache-dependency-path: FE/package-lock.json
      - run: npm ci
      - run: npx playwright install --with-deps chromium
      - run: npm run e2e
      - name: Rapporto e tracce se il giro fallisce
        if: failure()
        uses: actions/upload-artifact@v4
        with:
          name: playwright
          path: |
            FE/playwright-report
            FE/test-results
          retention-days: 7
```
